using System;
using System.Collections.Generic;
using MiniAdventureWorld.Core;
using MiniHero.ModApi;

namespace LocalAliceLanternTrailMod
{
    public sealed class LanternSight : TraitContent
    {
        public LanternSight() : base(ModEntry.PackageId + ".trait.lantern_sight") { }
        public override string Name => "Lantern Sight";
        public override string Description => "Basic attacks select one additional random enemy.";
        public override void OnBeforeNormalAttackTargets(BeforeNormalAttackTargetsEvent e)
        {
            if (e != null) e.additionalRandomTargets += 1;
        }
    }

    public sealed class SilverBolt : MoveContent, IModPresentationContent
    {
        public SilverBolt() : base(ModEntry.PackageId + ".move.silver_bolt") { }
        public override string Name => "Silver Bolt";
        public override string Description => "Deal damage equal to 2 times attack to one enemy.";
        public override MoveTargetKind TargetKind => MoveTargetKind.Enemy;
        public override int MaxUses => 2;
        public override int AttackMultiplier => 2;
        public ModPresentation Presentation => ModAssets.Icon("assets/icons/silver-bolt.png");
    }

    public sealed class LanternRanger : CharacterContent, IModPresentationContent
    {
        public LanternRanger() : base(ModEntry.PackageId + ".character.lantern_ranger") { }
        public override string Name => "Lantern Ranger";
        public override string Role => "Hunter";
        public override string School => "Woodland Watch";
        public override string Description => "Protects travelers with a lantern and light crossbow.";
        public override PlayerState BaseStats => new PlayerState
        { hp = 24, maxHp = 24, attack = 6, defense = 1, wisdom = 2 };
        public override IReadOnlyList<Type> TraitTypes => new[] { typeof(LanternSight) };
        public override IReadOnlyList<Type> StartingMoveTypes => new[] { typeof(SilverBolt) };
        // Current runtime limitation: StartingTreasureTypes is validated/stored but not granted on selection.
        // public override IReadOnlyList<Type> StartingTreasureTypes => new[] { typeof(LanternCompass) };
        public override IReadOnlyList<string> TreasurePoolTags => new[]
        { CharacterTreasurePoolTags.HealingLifeBuild, "treasure_pool.your_package.lantern" };
        // Optional: adjust the tag-derived pool for this character.
        // public override TreasurePoolDefinition TreasurePool => base.TreasurePool
        //     .Exclude("goblin_brass_scale")
        //     .Include(ModEntry.PackageId + ".treasure.lantern_compass")
        //     .SetWeight(ModEntry.PackageId + ".treasure.lantern_compass", 2f);
        public ModPresentation Presentation => ModAssets.Create(
            (ModPresentationSlot.Icon, "assets/icons/lantern-ranger.png"),
            (ModPresentationSlot.Portrait, "assets/icons/lantern-ranger.png"),
            (ModPresentationSlot.Avatar, "assets/portraits/lantern-ranger-avatar.png"));
    }
}
