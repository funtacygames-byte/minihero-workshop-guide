namespace LocalAliceLanternTrailMod

    public sealed class LanternSight : TraitContent
    {
        public LanternSight() : base(ModEntry.PackageId + ".trait.lantern_sight") { }
        public override string Name => ModText.Get(ModEntry.PackageId, "trait.lantern_sight.name", "灯辉视野");
        public override string Description => ModText.Get(ModEntry.PackageId, "trait.lantern_sight.description", "普通攻击额外选择一名随机敌人。");
        public override void OnBeforeNormalAttackTargets(BeforeNormalAttackTargetsEvent e)
        {
            if (e != null) e.additionalRandomTargets += 1;
        }
    }

    public sealed class SilverBolt : MoveContent

        public SilverBolt() : base(ModEntry.PackageId + ".move.silver_bolt") { }

        public override string Description => ModText.Get(ModEntry.PackageId, "move.silver_bolt.description", "对一名敌人造成 2 倍攻击伤害。");
        public override MoveTargetKind TargetKind => MoveTargetKind.Enemy;
        public override int MaxUses => 2;
        public override int AttackMultiplier => 2;
    }
