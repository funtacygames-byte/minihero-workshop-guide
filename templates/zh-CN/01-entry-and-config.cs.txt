    {
        public const string PackageId = "your.package";

        {

                "shield_per_stage", "每场景护盾", ModConfigValueKind.Integer,
                "2", "平衡", "进入场景时获得的护盾。", 0, 20));

            // 把其他模板中的实例放在这里：
